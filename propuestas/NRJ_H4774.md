### Campo observaciones_pedido
```sql
SELECT DISTINCT
  pc.fecha,
  pc.codigo,
  pc.idpedido AS id_pedido
FROM salidasalmacen sa
JOIN palets_mercado pm                    ON pm.id_salida = sa.id
JOIN lineaspedidoscli_palets_mercado lpm  ON lpm.id_palet_mercado = pm.id
JOIN pedidoscliproductoslinea ppl         ON ppl.id = lpm.id_linea
JOIN pedidoscli pc                        ON pc.idpedido = ppl.idpedido
WHERE sa.id = 29498          -- <-- id de la salida de almacén
ORDER BY pc.fecha, pc.codigo;
```
